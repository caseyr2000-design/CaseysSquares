  };
  const tok = () => store.get("sq_token"), adm = () => store.get("sq_admin");
  const run = (fn, ...args) => new Promise((resolve, reject) => {
    google.script.run
      .withSuccessHandler(resolve)
      .withFailureHandler((e) => reject(new Error(String((e && e.message) || e || "Something went wrong.").replace(/^(Exception|Error):\s*/, ""))))[fn](...args);
  });

  function withImages(st) {
    const urls = {};
    for (const [slot, ver] of Object.entries(st.cfg.images || {})) {
      if (imgVer[slot] === ver && imgData[slot]) urls[slot] = imgData[slot];
      else if (imgLoading[slot] !== ver) {
        imgLoading[slot] = ver;
        run("getImage", slot).then((uri) => {
          imgData[slot] = uri; imgVer[slot] = ver;
          if (last) emit(withImages(last));
        }).catch(() => { imgLoading[slot] = null; });
      }
    }
    return { ...st, cfg: { ...st.cfg, imageUrls: urls } };
  }

      const st = await run("getState", tok(), adm());
      if (!st.admin && adm()) store.del("sq_admin");
      last = st; emit(withImages(st));
    } catch (e) { emit(null, "Can't reach the board. Retrying…"); }
  }

  function fitCell(w, h) {
    const compact = isCompact(), rowHead = compact ? 0.6 : 0.8;
    const side = compact ? 0 : Math.max(30, Math.min(70, w * 0.04)) + 6;
    let c = (w - side - 26) / (10 + rowHead + EDGE);
    if (h) c = Math.min(c, (h - 70) / (10 + rowHead + EDGE + 0.7));
    return Math.floor(c);
  }

  function fitNames() {
    const brd = $("field").querySelector(".board");
    if (!brd || !(brd.classList.contains("lvl-i") || brd.classList.contains("lvl-s"))) return;
    for (const nm of brd.querySelectorAll(".cell.taken .nm")) {
      nm.style.fontSize = "";
      const cell = nm.parentElement, em = cell.querySelector(".em");
      const maxH = cell.clientHeight - (em ? em.offsetHeight : 0) - 3;
      let size = parseFloat(getComputedStyle(nm).fontSize), guard = 30;
      while (guard-- > 0 && size > 4 && (nm.scrollWidth > nm.clientWidth + 1.5 || nm.offsetHeight > maxH)) {
        size -= 0.5; nm.style.fontSize = size + "px";
      }
      // a very long single word on a tiny screen: let it wrap rather than be cut off
      nm.style.overflowWrap = nm.scrollWidth > nm.clientWidth + 1.5 ? "anywhere" : "";
    }
  }

  async function shrinkImage(file, slot) {
    if (!/^image\/(png|jpeg|webp)$/.test(file.type)) throw new Error("Use a PNG, JPG or WebP image.");
    if (file.size > 15 * 1024 * 1024) throw new Error("That image is over 15 MB. Pick a smaller one.");
    const url = URL.createObjectURL(file);
    try {
      const img = await new Promise((res, rej) => { const i = new Image(); i.onload = () => res(i); i.onerror = () => rej(new Error("Couldn't read that image.")); i.src = url; });
      const max = slot === "header" ? 1800 : 320, k = Math.min(1, max / Math.max(img.naturalWidth, img.naturalHeight));
      const cv = document.createElement("canvas"); cv.width = Math.round(img.naturalWidth * k); cv.height = Math.round(img.naturalHeight * k);
      cv.getContext("2d").drawImage(img, 0, 0, cv.width, cv.height);
      const type = slot === "header" ? "image/jpeg" : "image/png";
      return await new Promise((res, rej) => cv.toBlob((b) => (b ? res(b) : rej(new Error("Couldn't process that image."))), type, 0.85));
    } finally { URL.revokeObjectURL(url); }
  }

  // ---------- header banner ----------
  // Layout comes from config.js: SQUARES_HEADER_LAYOUT = { text: "left", clear: 0.19, bg: "#234c48" } puts the
  // title in the open space left of the picture's centre (clear = share of the picture's width that is empty on
  // each side); without it the title sits over the middle of the picture.

  // ---------- header title: three balanced lines, sized to fill the space over the logo ----------
  function setTitle(text) {
    const h = $("title"); if (h.dataset.t === text) return; h.dataset.t = text;
    const words = String(text || "").trim().split(/\s+/).filter(Boolean), n = Math.min(3, words.length);
    let best = null;
    // try every way to split the words into n lines; keep the one whose longest line is shortest
    const len = (a, b) => words.slice(a, b).join(" ").length;
    if (n <= 1) best = [words.join(" ")];
    else if (n === 2) { for (let i = 1; i < words.length; i++) { const m = Math.max(len(0, i), len(i)); if (!best || m < best.m) best = { m, l: [words.slice(0, i).join(" "), words.slice(i).join(" ")] }; } best = best.l; }
    else {
      for (let i = 1; i < words.length - 1; i++) for (let j = i + 1; j < words.length; j++) {
        const m = Math.max(len(0, i), len(i, j), len(j)); if (!best || m < best.m) best = { m, l: [words.slice(0, i), words.slice(i, j), words.slice(j)].map((w) => w.join(" ")) };
      }
      best = best.l;
    }
    h.replaceChildren(...best.map((t) => { const s = document.createElement("span"); s.className = "tl"; s.textContent = t; return s; }));
    h.setAttribute("aria-label", text);
  }
  function fitTitle() {
    const h = $("title"), hero = $("heroText"), banner = $("banner");
    h.style.fontSize = "";
    if (hero.parentElement !== banner) return;
    const lines = [...h.querySelectorAll(".tl")]; if (!lines.length) return;
    const cs = getComputedStyle(hero), W = hero.clientWidth - parseFloat(cs.paddingLeft) - parseFloat(cs.paddingRight);
    const other = [...hero.children].filter((c) => c !== h).reduce((a, c) => a + c.offsetHeight, 0);
    const H = banner.clientHeight * 0.98 - other - 12;
    h.style.fontSize = "100px";
    const w100 = Math.max(...lines.map((l) => l.offsetWidth)), h100 = h.offsetHeight;
    const size = Math.max(18, Math.min(220, Math.floor(Math.min(100 * W / w100, 100 * H / h100))));
    h.style.fontSize = size + "px";
  }
  function placeHero() {
    const hero = $("heroText"), banner = $("banner"), img = $("bannerImg");
    const L = window.SQUARES_HEADER_LAYOUT || null, usingDefault = !(S.st && S.st.cfg.imageUrls && S.st.cfg.imageUrls.header);
    const left = !!(L && L.text === "left" && usingDefault);
    banner.classList.toggle("layout-left", left);
    banner.style.background = left && L.bg ? L.bg : "";
    let wide = !banner.hidden && window.innerWidth >= 760;
    const between = !!(L && L.text === "between" && usingDefault);
    banner.classList.toggle("layout-between", between);
    $("band").style.background = (between || left) && L.strip ? L.strip : "";
    if (between) {
      // fill any space beside the picture with the colours at its left and right edges
      banner.style.background = "linear-gradient(90deg," + (L.bgLeft || "#000") + " 50%," + (L.bgRight || "#000") + " 50%)";
      const bw = banner.getBoundingClientRect(), iw = img.getBoundingClientRect();
      const x = iw.left - bw.left + iw.width * (L.from || 0.35), w = iw.width * ((L.to || 0.6) - (L.from || 0.35));
      banner.style.setProperty("--hx", Math.round(x) + "px"); banner.style.setProperty("--hw", Math.round(w) + "px");
      wide = !banner.hidden && w >= 230 && window.innerWidth >= 760;
    }
    if (left && !banner.hidden) {
      const W = banner.clientWidth, H = img.clientHeight || 1, aspect = img.naturalWidth && img.naturalHeight ? img.naturalWidth / img.naturalHeight : 2;
      const drawnW = Math.min(W, H * aspect), free = (W - drawnW) / 2 + drawnW * (L.clear || 0);
      banner.style.setProperty("--free", Math.floor(free) + "px");
      wide = free >= 280;
    }
    if (wide && hero.parentElement !== banner) banner.append(hero);
    else if (!wide && hero.parentElement === banner) $("band").querySelector(".wrap").prepend(hero);
    // In the left-title layout, the price/squares/pot figures fill the matching space on the right.
    // Otherwise they sit small in the banner's bottom-right corner, lined up with the picture's right edge.
    const stats = $("band").querySelector(".stats");
    const leftStats = left && wide, corner = !left && !banner.hidden, statsIn = leftStats || corner;
    if (statsIn && stats.parentElement !== banner) banner.append(stats);
    else if (!statsIn && stats.parentElement === banner) hero.append(stats);
    banner.classList.toggle("stats-corner", corner);
    if (corner) { const bw = banner.getBoundingClientRect(), iw = img.getBoundingClientRect(); banner.style.setProperty("--ir", Math.max(0, Math.round(bw.right - iw.right)) + "px"); }
    $("band").classList.toggle("stats-in-banner", leftStats);
    $("band").classList.toggle("hero-in-banner", hero.parentElement === banner);
    fitTitle();
  }
